#include <stdio.h>
#include <string.h>
#include <ctype.h>

char input[100];
int pos = 0;

int E(void);
int T(void);
int F(void);

int F(void) {
    if (input[pos] == '(') {
        pos++;
        if (E() && input[pos] == ')') {
            pos++;
            return 1;
        }
        return 0;
    }

    if (isalpha((unsigned char)input[pos])) {
        while (isalnum((unsigned char)input[pos]))
            pos++;
        return 1;
    }

    return 0;
}

int T(void) {
    if (!F())
        return 0;

    while (input[pos] == '*') {
        pos++;
        if (!F())
            return 0;
    }
    return 1;
}

int E(void) {
    if (!T())
        return 0;

    while (input[pos] == '+') {
        pos++;
        if (!T())
            return 0;
    }
    return 1;
}

int main(void) {
    scanf("%99s", input);

    if (E() && input[pos] == '\0')
        printf("String is accepted\n");
    else
        printf("String is not accepted\n");

    return 0;
}
